# ---------------------------
# Butterfly cache
# ---------------------------

# Lijst waarin de geladen vlinderafbeeldingen worden opgeslagen
butterfly_images = []

# Lijst met bron-URL's/base64 strings om dubbele afbeeldingen te voorkomen
butterfly_sources = []


def fetch_butterflies():
    """
    Haalt vlinders op uit de API en slaat ze lokaal op in de cache.
    Nieuwe afbeeldingen worden gedownload, verwerkt en toegevoegd
    aan butterfly_images.
    """
    
    global butterfly_images, butterfly_sources

    try:
        # Vraag data op van de API
        response = requests.get(API_URL, timeout=5)
        response.raise_for_status()

        # Zet JSON-response om naar Python-object
        data = response.json()

        added = 0

        # Verwerk maximaal de eerste 10 vlinders uit de API-response
        for b in data[:10]:

            # Zoek naar een afbeelding in verschillende mogelijke velden
            img_str = b.get("image") or b.get("img") or b.get("url")

            # Sla over als er geen afbeelding is
            # of als deze al in de cache zit
            if not img_str or img_str in butterfly_sources:
                continue

            try:
                # Controleer of de afbeelding base64-encoded is
                if img_str.startswith("data:image"):

                    # Decode base64 afbeelding naar bytes
                    img_data = base64.b64decode(img_str.split(",")[1])

                else:
                    # Download afbeelding vanaf URL
                    img_data = requests.get(img_str, timeout=5).content

                # Zet bytes om naar numpy-array
                img_array = np.frombuffer(img_data, np.uint8)

                # Decode afbeelding naar OpenCV-formaat
                img = cv2.imdecode(img_array, cv2.IMREAD_UNCHANGED)

                # Sla over als afbeelding niet correct geladen kon worden
                if img is None:
                    continue

                # Resize afbeelding naar gewenste vlindergrootte
                img = cv2.resize(img, (BUTTERFLY_SIZE, BUTTERFLY_SIZE))

                # Voeg afbeelding toe aan cache
                butterfly_images.append(img)

                # Bewaar bron zodat duplicaten voorkomen worden
                butterfly_sources.append(img_str)

                added += 1

                # Houd cachegrootte beperkt
                if len(butterfly_images) > MAX_CACHED_BUTTERFLIES:
                    butterfly_images.pop(0)
                    butterfly_sources.pop(0)

            except:
                # Sla foutieve afbeelding over en ga verder
                continue

        # Toon hoeveel nieuwe vlinders zijn toegevoegd
        if added:
            print(
                f"Fetched {added} new butterflies, "
                f"cache size: {len(butterfly_images)}"
            )

    except Exception as e:
        # Fout bij API-aanroep
        print("API error:", e)


# Start een achtergrondthread die continu nieuwe vlinders ophaalt
threading.Thread(

    # Oneindige loop:
    # 1. Haal nieuwe vlinders op
    # 2. Wacht API_FETCH_INTERVAL seconden
    target=lambda: [
        fetch_butterflies() or time.sleep(API_FETCH_INTERVAL)
        for _ in iter(int, 1)
    ],

    # Daemon=True zorgt ervoor dat de thread stopt wanneer het programma stopt
    daemon=True

).start()
