// ---------------------------
// Boss Defeat Tracking
// ---------------------------

// Dictionary that stores whether each boss has been defeated.
// The key is the boss name and the value is its defeat status.
[Header("Bosses Defeated")]
public Dictionary BossDefeated = new Dictionary()
{
    { "Bladder", false },
    { "Intestine", false },
    { "Liver", false },
    { "Lung", false },
    { "Throat", false }
};

// ---------------------------
// Boss Death Handling
// ---------------------------

public void Death(bool isDead)
{
    // Only execute the death logic if the boss has died.
    if (isDead)
    {
        // If this is the final boss, mark the game as completed.
        if (this.gameObject.name == "Brain Enemy")
        {
            PlayerHealth.Instance.GameBeat = true;
        }

        // Update the player's defeated boss list.
        CheckDefeatedBosses();

        // Spawn the reward after the boss dies if this component exists.
        SpawnAfterBossDeath spawnHeal = gameObject.GetComponent();

        if (spawnHeal != null)
        {
            spawnHeal.SpawnTheRedBloodCells();
        }

        // Remove the boss from the scene.
        Destroy(gameObject);

        Debug.Log("Boss is dead");
    }
}

// Checks which boss was defeated and updates its status
// in the player's BossDefeated dictionary.
public void CheckDefeatedBosses()
{
    switch (this.gameObject.name)
    {
        case "Bladder Enemy":
            UpdateBossStatus("Bladder");
            break;

        case "Intestines Enemy":
            UpdateBossStatus("Intestine");
            break;

        case "Liver Enemy":
            UpdateBossStatus("Liver");
            break;

        case "Lung Boss":
            UpdateBossStatus("Lung");
            break;

        case "Throat Enemy":
            UpdateBossStatus("Throat");
            break;
    }
}

// Marks the boss as defeated in the player's dictionary.
// This allows the game to remember that the boss has already
// been defeated and prevents it from respawning later.
private void UpdateBossStatus(string bossName)
{
    PlayerHealth.Instance.BossDefeated[bossName] = true;

    Debug.Log(bossName + " defeated value is now " +
              PlayerHealth.Instance.BossDefeated[bossName]);

    // Enable the exit arrow after defeating the boss.
    ExitArrow.SetActive(true);
}
