private void Update()
{
    _sceneName = SceneManager.GetActiveScene().name;
    //if you are in the digging scene
    if ((_sceneName == "fossils" || _sceneName == "rocks" ) )
    {
        //find the fossils/stones that have been spawned in
        _gemObj = GameObject.Find("gem");
        _gemScipt = _gemObj.GetComponent();

        //if a fossil has not been added yet
        if (_fossilAdded == false && _gemObj.activeInHierarchy)
        {
            //get the prefab of the stone/fossil
            FossilPrefabs.Add(_gemScipt.SelectedPrefab);
            _fossilAdded = true;
        }
    }
    else
    {
        //Once you've left the digging scene, set fossilAdded to false
        //This allows you to dig up another fossil later
        //and display it on the main digging screen
        _fossilAdded = false;
    }
    if(_sceneName == "SplashScreen")
    {
        //When you go back to the menu 
        //Remove all stone info
        DugUpRocks.Clear();
        NameOfRocks.Clear();
        FossilPrefabs.Clear();
    }

    if(_sceneName == "level 1")
    {
        
        //Find the excavated stones by name
        if(_rocksFound == false)
        {
            //Use the names to find the locations of the excavated stones
            foreach (string name in NameOfRocks)
            {
                DugUpRocks.Add(GameObject.Find(name));
            }
            _rocksFound = true;
        }
        //depending on which piles have already been dug up
        //spawn the correct fossil/rock prefab on each pile
        //then remove the piles
        for(int i = 0; i < DugUpRocks.Count; i++)
        {
            Instantiate(FossilPrefabs[i], DugUpRocks[i].transform.position, DugUpRocks[i].transform.rotation);
            Destroy(DugUpRocks[i]);
        }
        DugUpRocks.Clear();
    }
    else
    {
        //clean the list so there will not be any problems in lvl1
        RockChosen = false;
        DugUpRocks.Clear();
        _rocksFound= false;
    }
}
